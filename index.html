Provide SPREADSHEET_ID in Code.gs if using a standalone script.'); } return ss; } function doGet(e) { try { const params = (e && e.parameter) ? e.parameter : {}; const action = String(params.action || '').trim().toLowerCase(); const cb = params.callback || null; // 1. STUDENT SCORE MANAGEMENT if (action === 'delete_score') { return createJsonResponse(deleteStudentScore(params), cb); } // 3. STUDENT SCORE LOGGING if (action === 'log' || params.studentName) { const result = recordScoreToSheet({ timestamp: params.timestamp || new Date().toISOString(), studentName: params.studentName, studentSection: params.studentSection, classCode: params.classCode, categoryTitle: params.categoryTitle, submode: params.submode, configDetails: params.configDetails, score: params.score, total: params.total, percentage: params.percentage, mastery: params.mastery, questionDetails: params.questionDetails || '[]' }); return createJsonResponse(result, cb); } // 5. SERVE HTML return HtmlService.createHtmlOutputFromFile('Index') .setTitle('PROJECT PRIME | Multi-Teacher Diagnostic & Practice Suite') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no, viewport-fit=cover'); } catch (err) { return createJsonResponse({ status: 'error', message: err.toString() }, e?.parameter?.callback); } } function doPost(e) { try { let payload = {}; if (e && e.postData && e.postData.contents) { try { payload = JSON.parse(e.postData.contents); } if (payload.action === 'teacher_register') { return createJsonResponse(registerTeacher(payload)); } if (payload.action === 'teacher_update') { return createJsonResponse(updateTeacherProfile(payload)); } if (payload.action === 'teacher_delete') { return createJsonResponse(deleteTeacherAccount(payload)); } if (payload.action === 'delete_score') { return createJsonResponse(deleteStudentScore(payload)); } if (payload.action === 'fetch') { const auth = loginTeacher(payload.username, payload.password); if (auth.status !== 'success') { return createJsonResponse({ status: 'unauthorized', message: 'Access denied.', records: [] }); } const records = queryScoresFromSheet(auth.teacher.classCode); return createJsonResponse({ status: 'success', count: records.length, records: records }); } const result = recordScoreToSheet(payload); } function registerTeacher(data) { const ss = getSpreadsheet(); let sheet = ss.getSheetByName(TEACHERS_SHEET); if (!sheet) { sheet = ss.insertSheet(TEACHERS_SHEET); const headers = ['Created At', 'Full Name', 'Username / ID', 'Password Hash', 'Class Code', 'Sections']; sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setBackground('#0f172a').setFontColor('#fbbf24').setFontWeight('bold'); sheet.setFrozenRows(1); sheet.setRowHeight(1, 36); } const username = String(data.username || '').trim().toUpperCase(); const password = String(data.password || '').trim(); const fullName = String(data.fullName || '').trim(); const classCode = String(data.classCode || username).trim().toUpperCase(); const sections = String(data.sections || '').trim(); if (!username || !password) { return { status: 'error', message: 'Username and password are required.' }; } const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][2]).toUpperCase() === username) { return { status: 'error', message: 'Teacher ID / Username already exists.' }; } } sheet.appendRow([new Date(), fullName, username, password, classCode, sections]); return { status: 'success', teacher: { fullName: fullName, username: username, classCode: classCode, sections: sections } }; } function loginTeacher(username, password) { const ss = getSpreadsheet(); const sheet = ss.getSheetByName(TEACHERS_SHEET); if (!sheet) return { status: 'error', message: 'No registered teacher database found.' }; const targetUser = String(username || '').trim().toUpperCase(); const targetPass = String(password || '').trim(); if (!targetUser || !targetPass) { return { status: 'error', message: 'Username and password cannot be empty.' }; } const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { const rowUser = String(rows[i][2] || '').trim().toUpperCase(); const rowPass = String(rows[i][3] || '').trim(); if (rowUser === targetUser && rowPass === targetPass) { return { status: 'success', teacher: { fullName: String(rows[i][1] || ''), username: rowUser, classCode: String(rows[i][4] || rowUser).trim().toUpperCase(), sections: String(rows[i][5] || '') } }; } } return { status: 'error', message: 'Invalid Teacher Username or Password.' }; } function updateTeacherProfile(data) { const ss = getSpreadsheet(); const sheet = ss.getSheetByName(TEACHERS_SHEET); if (!sheet) return { status: 'error', message: 'Teacher database not found.' }; const targetUser = String(data.username || '').trim().toUpperCase(); const currentPass = String(data.currentPassword || data.password || '').trim(); const auth = loginTeacher(targetUser, currentPass); if (auth.status !== 'success') { return { status: 'unauthorized', message: 'Password verification failed.' }; } const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][2] || '').trim().toUpperCase() === targetUser) { const rowIndex = i + 1; const newFullName = data.fullName !== undefined ? String(data.fullName).trim() : rows[i][1]; const newPassword = data.newPassword && String(data.newPassword).trim() !== '' ? String(data.newPassword).trim() : rows[i][3]; const newClassCode = data.classCode ? String(data.classCode).trim().toUpperCase() : rows[i][4]; const newSections = data.sections !== undefined ? String(data.sections).trim() : rows[i][5]; sheet.getRange(rowIndex, 2).setValue(newFullName); sheet.getRange(rowIndex, 4).setValue(newPassword); sheet.getRange(rowIndex, 5).setValue(newClassCode); sheet.getRange(rowIndex, 6).setValue(newSections); return { status: 'success', message: 'Profile updated successfully.', teacher: { fullName: newFullName, username: targetUser, classCode: newClassCode, sections: newSections } }; } } return { status: 'error', message: 'Teacher record not found.' }; } function deleteTeacherAccount(data) { const ss = getSpreadsheet(); const teacherSheet = ss.getSheetByName(TEACHERS_SHEET); if (!teacherSheet) return { status: 'error', message: 'Teacher database not found.' }; const targetUser = String(data.username || '').trim().toUpperCase(); const password = String(data.password || '').trim(); const auth = loginTeacher(targetUser, password); if (auth.status !== 'success') { return { status: 'unauthorized', message: 'Authentication failed. Incorrect password.' }; } const targetCode = auth.teacher.classCode; const rows = teacherSheet.getDataRange().getValues(); let found = false; for (let i = 1; i < rows.length; i++) { if (String(rows[i][2] || '').trim().toUpperCase() === targetUser) { teacherSheet.deleteRow(i + 1); found = true; break; } } if (!found) { return { status: 'error', message: 'Teacher account could not be found.' }; } const scoreSheet = ss.getSheetByName(SCORES_SHEET); if (scoreSheet) { const scoreData = scoreSheet.getDataRange().getValues(); for (let j = scoreData.length - 1; j >= 1; j--) { const code = String(scoreData[j][3] || '').trim().toUpperCase(); if (code === targetCode || code.startsWith(targetCode + '-')) { scoreSheet.deleteRow(j + 1); } } } return { status: 'success', message: 'Teacher profile and corresponding score records removed.' }; } function deleteStudentScore(data) { const ss = getSpreadsheet(); const scoreSheet = ss.getSheetByName(SCORES_SHEET); if (!scoreSheet) return { status: 'error', message: 'Score registry not found.' }; const auth = loginTeacher(data.username, data.password); if (auth.status !== 'success') { return { status: 'unauthorized', message: 'Access denied. Password verification failed.' }; } const teacherCode = auth.teacher.classCode; const targetTime = data.timestamp ? new Date(data.timestamp).getTime() : 0; const targetStudent = String(data.studentName || '').trim().toUpperCase(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const rows = scoreSheet.getDataRange().getValues(); // Loop backwards to safely delete rows without affecting indices of yet-to-be-checked rows for (let i = rows.length - 1; i >= 1; i--) { const rowCode = String(rows[i][3] || '').trim().toUpperCase(); const rowStudent = String(rows[i][1] || '').trim().toUpperCase(); const cellTime = rows[i][0] ? new Date(rows[i][0]).getTime() : 0; const matchCode = rowCode === teacherCode || rowCode.startsWith(teacherCode + '-'); const matchTime = Math.abs(cellTime - targetTime) < 5000; if (matchCode && rowStudent === targetStudent && matchTime) { scoreSheet.deleteRow(i + 1); return { status: 'success', message: 'Student record deleted successfully.' }; } } return { status: 'not_found', message: 'Record not found or already deleted.' }; } finally { lock.releaseLock(); } } function recordScoreToSheet(data) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const ss = getSpreadsheet(); let sheet = ss.getSheetByName(SCORES_SHEET); if (!sheet) { sheet = ss.insertSheet(SCORES_SHEET); const headers = [ 'Timestamp', 'Student Name', 'Grade & Section', 'Teacher Class Code', 'Category', 'Submode / Operation', 'Configuration Details', 'Score', 'Total Items', 'Accuracy %', 'Mastery Rating', 'Question Details' ]; sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length) .setBackground('#0f172a').setFontColor('#fbbf24').setFontWeight('bold').setHorizontalAlignment('center'); sheet.setFrozenRows(1); sheet.setRowHeight(1, 36); } let pctDisplay = '0%'; if (data.percentage !== undefined && data.percentage !== null && data.percentage !== '') { pctDisplay = String(data.percentage).replace('%', '') + '%'; } else if (data.score !== undefined && data.total) { pctDisplay = Math.round((Number(data.score) / Number(data.total)) * 100) + '%'; } const entryDate = data.timestamp ? new Date(data.timestamp) : new Date(); sheet.appendRow([ entryDate, data.studentName || 'Anonymous Student', data.studentSection || 'General', (data.classCode || 'GENERAL').toUpperCase(), data.categoryTitle || data.categoryId || 'General Math', data.submode || 'Standard', data.configDetails || '', Number(data.score) || 0, Number(data.total) || 0, pctDisplay, data.mastery || 'Completed', typeof data.questionDetails === 'string' ? data.questionDetails : JSON.stringify(data.questionDetails || []) ]); const lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 4, 1, 1).setHorizontalAlignment('center'); sheet.getRange(lastRow, 8, 1, 3).setHorizontalAlignment('center'); return { status: 'success', row: lastRow }; } finally { lock.releaseLock(); } } function queryScoresFromSheet(classCode) { const ss = getSpreadsheet(); const sheet = ss.getSheetByName(SCORES_SHEET); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const target = String(classCode || '').trim().toUpperCase(); if (!target || target === 'ALL') { return []; } const results = []; for (let i = 1; i < data.length; i++) { const row = data[i]; const rowCode = String(row[3] || '').trim().toUpperCase(); if (rowCode === target || rowCode.startsWith(target + '-')) { results.push({ id: 'cloud_row_' + i + '_' + (row[0] ? new Date(row[0]).getTime() : Date.now()), timestamp: row[0] ? new Date(row[0]).toISOString() : new Date().toISOString(), studentName: String(row[1] || ''), studentSection: String(row[2] || ''), classCode: rowCode, categoryTitle: String(row[4] || ''), submode: String(row[5] || ''), configDetails: String(row[6] || ''), score: Number(row[7]) || 0, total: Number(row[8]) || 0, percentage: parseInt(String(row[9]).replace(/[^\d]/g, ''), 10) || 0, mastery: String(row[10] || 'Completed'), questionDetails: row[11] ? String(row[11]) : '[]' }); } } return results.reverse().slice(0, 250); }
